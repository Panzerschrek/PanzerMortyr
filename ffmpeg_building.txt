Install msys2


Install "make" within msys2 shell:
  pacman -S make

Run cmd

Type
  call "C:\Program Files (x86)\Microsoft Visual Studio\2019\Professional\VC\Auxiliary\Build\vcvarsall.bat" amd64

Run msys2 shell from cmd:
  msys2_shell.cmd -use-full-path

Configure:
  ./configure --target-os=win64 --arch=x86_64 --toolchain=msvc --disable-x86asm --disable-static --enable-shared --disable-programs --disable-doc --disable-everything --disable-hwaccels --disable-swscale --enable-decoder=indeo3 --enable-decoder=indeo5 --enable-muxer=avi

Build:
  make -j 4
